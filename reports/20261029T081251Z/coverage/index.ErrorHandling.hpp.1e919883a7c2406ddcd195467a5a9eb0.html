4 /*
5 The ErrorHandling class handles all errors from BSP/ConnectorLock etc
6 and classifies them for charging:
7
8 If we can continue charging despite the error, we treat them as warnings and do not track them here. If e.g. the BSP
9 raises a high temperature error it needs to limit the output power by itself.
10
11 The decision whether an error requires a replug or not to clear depends on the reporting module. It will need to clear
12 them at the appropriate time.
13 */

111 // Orders errors by (type, sub_type) only, so inoperative_causes tracks the identity of the active fatal set
112 // regardless of detection order.
113 struct ErrorCauseLess {
114 4 bool operator()(const Everest::error::Error& a, const Everest::error::Error& b) const {
115 4 return std::tie(a.type, a.sub_type) < std::tie(b.type, b.sub_type);
116 }
117 };
