99 class MonitoringUpdater {
100
101 public:
102 MonitoringUpdater() = delete;
103
104 /// \brief Constructs a new variable monitor updater
105 /// \param device_model Currently used variable device model
106 /// \param notify_csms_events Function that can be invoked with a number of alert events
107 /// \param is_chargepoint_offline Function that can be invoked in order to retrieve the
108 /// status of the charging station connection to the CSMS
109 MonitoringUpdater(DeviceModelMonitoringInterface& device_model, notify_events notify_csms_events,
110 is_offline is_chargepoint_offline);
111 ~MonitoringUpdater();
112
113 /// \brief Starts monitoring the variables, kicking the timer
114 void start_monitoring();
115 /// \brief Stops monitoring the variables, canceling the timer
116 void stop_monitoring();
117
118 /// \brief Processes the variable triggered monitors. Will be called
119 /// after relevant variable modification operations or will be called
120 /// periodically in case that processing can not be done at the current
121 /// moment, for example in the case of an internal variable modification
122 void process_triggered_monitors();
123
124 private:
125 /// \brief Monitor state shared by the monitoring timer and the device model listeners
126 struct State {
127 // Charger to CSMS message unique ID for EventData
128 std::int32_t unique_id{0};
129 std::unordered_map<std::int32_t, UpdaterMonitorMeta> updater_monitors_meta;
130 };
131
132 /// \brief Callback that is registered to the 'device_model' that determines if any of
133 /// the monitors are triggered for a certain variable when the internal value is used. Will
134 /// delay the sending of the monitors to the CSMS until the charging station has
135 /// finished any current operation. The reason is that a variable can change during an
136 /// operation where the CSMS does NOT expect a message of type 'EventData' therefore
137 /// the processing is delayed either until a manual call to 'process_triggered_monitors'
138 /// or when the periodic monitoring timer is hit
139 void on_variable_changed(const std::unordered_map<std::int64_t, VariableMonitoringMeta>& monitors,
140 const Component& component, const Variable& variable,
141 const VariableCharacteristics& characteristics, const VariableAttribute& attribute,
142 const std::string& value_previous, const std::string& value_current);
143
144 /// \brief Callback that is registered to the 'device_model' that determines if any of
145 /// the already existing monitors were updated. It is required for some spec requirements
146 /// that must refresh monitor data in the case of a monitor update
147 void on_monitor_updated(const VariableMonitoringMeta& updated_monitor, const Component& component,
148 const Variable& variable, const VariableCharacteristics& characteristics,
149 const VariableAttribute& attribute, const std::string& current_value);
150
151 /// \brief Evaluates if an monitor was triggered, and if it is triggered
152 /// it adds it to our internal list
153 void evaluate_monitor(State& state, const VariableMonitoringMeta& monitor_meta, const Component& component,
154 const Variable& variable, const VariableCharacteristics& characteristics,
155 const VariableAttribute& attribute, const std::string& value_previous,
156 const std::string& value_current);
157
158 /// \brief Processes the periodic monitors. Since this can be somewhat of a costly
159 /// operation (DB query of each triggered monitor's actual value) the processing time
160 /// can be configured using the 'VariableMonitoringProcessTime' internal variable. If
161 // there are also any pending alert triggered monitors, those will be processed too
162 void process_monitors_internal(bool allow_periodics, bool allow_trigger);
163
164 /// \brief Processes the monitor meta, generating in it's internal list all the
165 /// required events. It will generate the EventData for a notify regardless
166 /// of the offline state
167 void process_monitor_meta_internal(State& state, UpdaterMonitorMeta& updater_meta_data);
168
169 /// \brief Moves the reference of a Delta monitor to the value of an event that is sent or queued

171 void update_delta_reference(std::int32_t monitor_id, const std::string& reference_value);
172
173 /// \brief Updates our internal monitors with the periodic monitors of the device model
174 void update_periodic_monitors_internal(State& state);
