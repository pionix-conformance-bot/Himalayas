29 // Bits without a matching Enable in DERControlCPDRes: bits 0 and 1 (charge, discharge) are inherent to the
30 // service and bits 21 and 22 (charge loop target powers) carry no Enable, so neither group is ever enabled by
31 // the SECC.
32 inline constexpr std::uint32_t SAE_NOT_ENABLEABLE_MASK =
33 sae_function_bit(DerBitMapFunctions::ChargeFunction) | sae_function_bit(DerBitMapFunctions::DischargeFunction) |

67 constexpr bool is_sae_function_enableable(DerBitMapFunctions function) {
68 return (sae_function_bit(function) & SAE_NOT_ENABLEABLE_MASK) == 0;
69 }
70
71 constexpr std::size_t count_sae_enableable_functions() {
72 std::size_t count = 0;
73 for (const auto& entry : SAE_FUNCTION_NAMES) {
74 if (is_sae_function_enableable(entry.function)) {
75 ++count;
76 }
77 }
78 return count;
79 }
80
81 constexpr auto make_sae_enableable_functions() {
82 std::array<SaeFunctionName, count_sae_enableable_functions()> functions{};
83 std::size_t index = 0;
84 for (const auto& entry : SAE_FUNCTION_NAMES) {
85 if (is_sae_function_enableable(entry.function)) {
86 functions[index++] = entry;
87 }
88 }
89 return functions;
90 }
